jug1 = 0
jug2 = 0

while jug1 != 2:

    # Fill Jug 1
    if jug1 == 0:
        jug1 = 4
        print("Fill Jug 1")

    # Pour Jug 1 into Jug 2
    elif jug2 < 3:
        amount = min(jug1, 3 - jug2)
        jug1 = jug1 - amount
        jug2 = jug2 + amount
        print("Pour Jug 1 into Jug 2")

    # Empty Jug 2
    else:
        jug2 = 0
        print("Empty Jug 2")

    print("Jug 1 =", jug1, "Jug 2 =", jug2)

print("Goal Achieved!")
print("2 liters of water is in Jug 1.")
