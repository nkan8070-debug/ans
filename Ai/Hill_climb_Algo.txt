def f(x):
    return -(x - 5) ** 2 + 25

def hill_climbing(start):
    current = start
    while True:
        print("Current:", current, "Value:", f(current))
        right, left = f(current + 1), f(current - 1)
        if right > f(current) and right >= left:
            current += 1
        elif left > f(current):
            current -= 1
        else:
            break
    return current

start = int(input("Enter starting value: "))
result = hill_climbing(start)
print("Best value:", result)
print("Maximum value:", f(result))
