def hill_climbing(start):
    current = start

    while True:
        current_value = -(current - 5) ** 2 + 25
        next_value = -(current + 1 - 5) ** 2 + 25

        print("Current:", current, "Value:", current_value)

        if next_value > current_value:
            current = current + 1
        else:
            break

    return current


start = int(input("Enter starting value: "))

result = hill_climbing(start)

print("Best value:", result)
print("Maximum value:", -(result - 5) ** 2 + 25)
