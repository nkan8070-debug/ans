def alpha_beta(depth, node, alpha, beta, maximizing):
    
    # Heuristic value at depth limit
    if depth == 0:
        return node

    if maximizing:
        best = -999

        for value in [node - 2, node + 2]:
            score = alpha_beta(depth - 1, value, alpha, beta, False)
            best = max(best, score)
            alpha = max(alpha, best)

            if alpha >= beta:
                break

        return best

    else:
        best = 999

        for value in [node - 2, node + 2]:
            score = alpha_beta(depth - 1, value, alpha, beta, True)
            best = min(best, score)
            beta = min(beta, best)

            if alpha >= beta:
                break

        return best


# Input
start = int(input("Enter starting value: "))
depth = int(input("Enter depth limit: "))

# Alpha-Beta
result = alpha_beta(depth, start, -999, 999, True)

print("Best value:", result)
