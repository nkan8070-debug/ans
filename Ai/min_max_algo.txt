def minimax(depth, node, maximizing):

    # If we reach the end, return the value
    if depth == 0:
        return node

    if maximizing:
        best = -999

        for value in [node - 2, node + 2]:
            score = minimax(depth - 1, value, False)
            best = max(best, score)

        return best

    else:
        best = 999

        for value in [node - 2, node + 2]:
            score = minimax(depth - 1, value, True)
            best = min(best, score)

        return best


start = int(input("Enter starting value: "))
depth = int(input("Enter depth: "))

result = minimax(depth, start, True)

print("Best value:", result)
